% Slide 10 — How preservation is calculated
% ============================================================================
\begin{frame}{How is Preservation Calculated?}
\begin{columns}[T]
  \begin{column}{0.50\textwidth}
    \begin{block}{Symmetric preservation ratio}
      \scriptsize
      Each graph metric is first converted into a preservation score using the symmetric ratio
      \[\mathrm{Preservation\,(\%)} =\frac{\min(\mathrm{Baseline},\mathrm{Perturbed})}{\max(\mathrm{Baseline},\mathrm{Perturbed})}\times 100\]
      where \emph{Baseline} = original graph metric and \emph{Perturbed} = graph metric after reconstruction error.

      \medskip
      Category preservation is then computed as the \emph{unweighted arithmetic mean} of all preservation metrics belonging to that biological category.
    \end{block}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{block}{Example --- category $=$ Average(member metrics)}
      \scriptsize
      Structural Topology $=$ Average(Node Count, Edge Count, Density)\\
      Synaptic Properties $=$ Average(Total Synapses, Weight Mean, Weight Median,\\
      \hspace{1.15cm} Weight Variance, Weight Standard Deviation,\\
      \hspace{1.15cm} Weight Maximum, Weight Minimum)\\
      Connectivity $=$ Average(Largest WCC, Largest SCC)\\
      Network Organization $=$ Average(Reciprocity, Degree Assortativity)
    \end{block}
  \end{column}
\end{columns}

\vspace{-3pt}
\noindent\begin{beamercolorbox}[rounded=true,sep=2pt]{block body}
  {\scriptsize\color{navy}\textbf{Scientific Note:}\ Rather than combining heterogeneous graph metrics into one overall preservation score, the report summarizes preservation separately for each biological category. This provides a more interpretable view of how reconstruction errors affect different aspects of the connectome while preserving the original metric-level calculations.}
\end{beamercolorbox}
\end{frame}

% Slide 11 — Conclusions
% ============================================================================
\begin{frame}{Conclusions}
\begin{block}{Direct, measurable effects}
  \footnotesize Missed synapses remove synapses \emph{exactly} in proportion to the rate ($-20.0\%$ at 20\%); false synapses add edges \emph{exactly} in proportion ($+20.0\%$). These are the most sensitive quantities measured.
\end{block}
\begin{block}{Topology is highly robust}
  \footnotesize The giant connected component and node count survive to $>$99\% even at 20\% error --- the coarse wiring plan is barely disturbed (PageRank correlation stays $\ge$0.98).
\end{block}
\begin{block}{False synapses are the more disruptive model}
  \footnotesize At equal nominal rates they distort degree distributions 14--27$\times$ more (in-degree KS), erode hub ranking (top-$k$ 0.89 vs 0.97 for missed) and raise reciprocity by $+7\%$.
\end{block}
\begin{block}{Error signatures are separable}
  \footnotesize Missed synapses \emph{deflate} synaptic statistics; false synapses \emph{inflate} the graph --- weak edges ($\approx$3 synapses) dilute weights without creating strong hubs.
\end{block}
\end{frame}
